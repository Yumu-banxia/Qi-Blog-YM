代码块占位

const cache = new Map();

function memoize(fn) {
  return function (...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) {
      return cache.get(key);
    }
    const value = fn.apply(this, args);
    cache.set(key, value);
    return value;
  };
}

const slow = (n) => {
  let total = 0;
  for (let i = 0; i < n; i += 1) {
    total += Math.sqrt(i);
  }
  return total;
};

const fast = memoize(slow);
console.log(fast(100000));
